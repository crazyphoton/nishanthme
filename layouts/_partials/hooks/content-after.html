{{- /* Prev/next navigation at the foot of each post. Injected via Ananke's "content-after" hook (called from single.html), so no template override. Older/newer are decided by comparing .Date rather than relying on Hugo's .PrevInSection / .NextInSection naming, which is easy to get backwards. */ -}} {{- if eq .Section "posts" -}} {{- $cur := . -}} {{- $older := false -}} {{- $newer := false -}} {{- range (slice .PrevInSection .NextInSection) -}} {{- with . -}} {{- if .Date.Before $cur.Date -}} {{- $older = . -}} {{- else -}} {{- $newer = . -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $older $newer -}}
{{- with $older -}} ← Older {{ .Title }} {{- end -}}
{{- with $newer -}} Newer → {{ .Title }} {{- end -}}
{{- end -}} {{- end -}}
